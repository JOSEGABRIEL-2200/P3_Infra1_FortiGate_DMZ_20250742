! ---------- PARTE 1: basico y VLANs ----------
enable
configure terminal
hostname SW-A
no ip domain-lookup
ip routing
vlan 10
 name USUARIOS
vlan 20
 name ADMINISTRACION
vlan 99
 name ENLACE-FORTIGATE
vlan 999
 name SIN-USO
exit

! ---------- PARTE 2: puertos ----------
interface Ethernet0/0
 description ENLACE-FGT-DMZ-port2
 switchport mode access
 switchport access vlan 99
 spanning-tree portfast edge
 no shutdown
interface Ethernet0/1
 description USUARIO-VLAN10 (Windows 10 - Cloud2)
 switchport mode access
 switchport access vlan 10
 switchport port-security maximum 3
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 switchport port-security
 spanning-tree portfast edge
 spanning-tree bpduguard enable
 no shutdown
interface Ethernet0/2
 description USUARIO-VLAN20 (PC administrador - Cloud4)
 switchport mode access
 switchport access vlan 20
 switchport port-security maximum 3
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 switchport port-security
 spanning-tree portfast edge
 spanning-tree bpduguard enable
 no shutdown
interface Ethernet0/3
 description SIN-USO
 switchport mode access
 switchport access vlan 999
 shutdown
exit

! ---------- PARTE 3: gateways (SVI), ruta y DHCP ----------
interface Vlan10
 description GATEWAY-VLAN10-USUARIOS
 ip address 10.7.42.1 255.255.255.128
 no shutdown
interface Vlan20
 description GATEWAY-VLAN20-ADMINISTRACION
 ip address 10.7.42.129 255.255.255.128
 no shutdown
interface Vlan99
 description ENLACE-FORTIGATE
 ip address 10.7.43.18 255.255.255.252
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 10.7.43.17
ip dhcp excluded-address 10.7.42.1 10.7.42.9
ip dhcp excluded-address 10.7.42.129 10.7.42.137
ip dhcp pool VLAN10-USUARIOS
 network 10.7.42.0 255.255.255.128
 default-router 10.7.42.1
 dns-server 10.7.43.17
ip dhcp pool VLAN20-ADMINISTRACION
 network 10.7.42.128 255.255.255.128
 default-router 10.7.42.129
 dns-server 10.7.43.17
exit

! ---------- PARTE 4: seguridad basica ----------
service password-encryption
enable secret cisco123
banner motd #SW-A - Acceso restringido. P3 Infra 1 - 2025-0742#
ip domain-name itla.lab
username admin privilege 15 secret cisco123
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
access-list 20 remark Solo la VLAN 20 (administracion) administra el switch
access-list 20 permit 10.7.42.128 0.0.0.127
line console 0
 password cisco
 login
 logging synchronous
line vty 0 4
 access-class 20 in
 login local
 transport input ssh
end
write memory
